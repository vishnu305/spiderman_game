const GROQ = 'https://api.groq.com/openai/v1';
function base(cfg) { return cfg.worker ? cfg.worker.replace(/\/$/, '') + '/groq' : GROQ; }
function headers(cfg, json = true) { const h = {}; if (cfg.key) h.Authorization = 'Bearer ' + cfg.key; if (json) h['Content-Type'] = 'application/json'; return h; }
async function check(r) { if (!r.ok) { let t = ''; try { t = (await r.text()).slice(0, 200); } catch (e) {} throw new Error(`Groq ${r.status}: ${t}`); } return r; }

export async function chat(messages, cfg, { json = false, temperature = 0.6, max_tokens = 700 } = {}) {
  const models = [cfg.model, 'llama-3.1-8b-instant'];     // automatic fallback model
  let err;
  for (const model of models) {
    try {
      const body = { model, messages, temperature, max_tokens };
      if (json) body.response_format = { type: 'json_object' };
      const r = await check(await fetch(base(cfg) + '/chat/completions', { method: 'POST', headers: headers(cfg), body: JSON.stringify(body) }));
      return (await r.json()).choices[0].message.content.trim();
    } catch (e) { err = e; }
  }
  throw err;
}
export async function transcribe(blob, cfg) {
  const f = new FormData(); f.append('file', blob, 'audio.webm'); f.append('model', cfg.stt || 'whisper-large-v3-turbo'); f.append('response_format', 'json');
  const r = await check(await fetch(base(cfg) + '/audio/transcriptions', { method: 'POST', headers: headers(cfg, false), body: f }));
  return ((await r.json()).text || '').trim();
}

export async function writeBulletin(items, cfg) {
  const list = items.map((s, i) => `${i + 1}. ${s.title} — ${s.desc} (source: ${s.source})`).join('\n');
  const prompt = `${list}\n\nWrite a spoken bulletin as JSON: {"intro":"1 short greeting sentence","stories":["1-2 sentence spoken script per story, same order, mention source name once in a while"],"outro":"1 short sign-off with the line 'stay webbed'"}`;
  const txt = await chat([{ role: 'system', content: PERSONA }, { role: 'user', content: prompt }], cfg, { json: true, temperature: 0.5 });
  const j = JSON.parse(txt);
  if (!Array.isArray(j.stories) || j.stories.length < items.length) throw new Error('Bad bulletin JSON');
  return j;
}

const STOP = new Set('a an the and or of to in on for with is are was were be by at from as that this it its about what who when where why how tell me news latest today any'.split(' '));
const tok = (s) => (s.toLowerCase().match(/[a-z0-9\u0900-\u0dff]+/g) || []).filter((w) => !STOP.has(w));
let IDX = { docs: [], df: new Map(), avg: 1 };
export function buildIndex(items) {
  const docs = items.map((it) => { const t = tok(it.title + ' ' + it.title + ' ' + it.desc + ' ' + it.source); const tf = new Map(); t.forEach((w) => tf.set(w, (tf.get(w) || 0) + 1)); return { it, tf, len: t.length }; });
  const df = new Map(); docs.forEach((d) => d.tf.forEach((_, w) => df.set(w, (df.get(w) || 0) + 1)));
  IDX = { docs, df, avg: docs.reduce((a, d) => a + d.len, 0) / Math.max(1, docs.length) };
}
export function search(q, k = 5) {
  const N = IDX.docs.length, k1 = 1.5, b = 0.75, qs = tok(q);
  return IDX.docs.map((d) => {
    let s = 0;
    for (const w of qs) { const f = d.tf.get(w) || 0; if (!f) continue; const n = IDX.df.get(w); const idf = Math.log(1 + (N - n + 0.5) / (n + 0.5)); s += idf * (f * (k1 + 1)) / (f + k1 * (1 - b + b * d.len / IDX.avg)); }
    return { it: d.it, s };
  }).filter((x) => x.s > 0).sort((a, b) => b.s - a.s).slice(0, k).map((x) => x.it);
}

export async function fetchNews(topic, cfg) {
  if (QS.get('demo')) return DEMO;
  if (cfg.worker) {
    const r = await fetch(`${cfg.worker.replace(/\/$/, '')}/news?topic=${topic}`);
    if (!r.ok) throw new Error('News worker error ' + r.status);
    return (await r.json()).items;
  }
  const out = [];
  try { // Wikipedia current events (free, CORS ok)
    const d = new Date(); const p = (n) => String(n).padStart(2, '0');
    const r = await fetch(`https://api.wikimedia.org/feed/v1/wikipedia/en/featured/${d.getUTCFullYear()}/${p(d.getUTCMonth() + 1)}/${p(d.getUTCDate())}`);
    const j = await r.json();
    (j.news || []).forEach((n) => out.push({ title: strip(n.story).slice(0, 160), desc: strip(n.story), source: 'Wikipedia In the News', url: n.links?.[0]?.content_urls?.desktop?.page || '#', time: Date.now() }));
  } catch (e) { /* ignore */ }
  try { // Hacker News top stories (tech)
    const ids = await (await fetch('https://hacker-news.firebaseio.com/v0/topstories.json')).json();
    const top = await Promise.all(ids.slice(0, 12).map((id) => fetch(`https://hacker-news.firebaseio.com/v0/item/${id}.json`).then((r) => r.json())));
    top.filter(Boolean).forEach((s) => out.push({ title: s.title, desc: s.title, source: 'Hacker News', url: s.url || '#', time: (s.time || 0) * 1000 }));
  } catch (e) { /* ignore */ }
  if (!out.length) throw new Error('No news source reachable. Deploy the worker (see docs) or check internet.');
  return out;
}

function pick(name, lang) {
  if (name) { const v = voices.find((v) => v.name === name); if (v) return v; }
  const same = voices.filter((v) => v.lang.toLowerCase().startsWith(lang.slice(0, 2).toLowerCase()));
  const male = /(male|guy|ravi|david|mark|daniel|alex|james|george|rishi|aaron|fred|prabhat|madhur)/i, fem = /(female|zira|heera|neerja|samantha|susan|hazel|swara)/i;
  return same.find((v) => male.test(v.name) && !fem.test(v.name)) || same[0] || voices[0] || null;
}
// speak(): returns a promise that resolves when finished; onBeat is called on every word (used for lip-sync / head bob)
export function speak(text, { voice = '', lang = 'en-IN', rate = 1.02, pitch = 0.95, onBeat = () => {} } = {}) {
  return new Promise((resolve) => {
    if (!('speechSynthesis' in window)) return resolve(false);
    speechSynthesis.cancel();
    const parts = (text.match(/[^.!?;\n]+[.!?;]*/g) || [text]).map((s) => s.trim()).filter(Boolean); let i = 0;
    const next = () => {
      if (i >= parts.length) return resolve(true);
      const u = new SpeechSynthesisUtterance(parts[i++]), v = pick(voice, lang);
      if (v) { u.voice = v; u.lang = v.lang; } else u.lang = lang;
      u.rate = rate; u.pitch = pitch; u.onboundary = onBeat; u.onend = next; u.onerror = next; speechSynthesis.speak(u);
    };
    next();
  });
}

function wrap(g, text, x, y, maxW, lh, maxLines) {
  const words = (text || '').split(/\s+/); let line = '', n = 0;
  for (let i = 0; i < words.length; i++) {
    const t = line ? line + ' ' + words[i] : words[i];
    if (g.measureText(t).width > maxW && line) { g.fillText(line, x, y); y += lh; line = words[i]; if (++n >= maxLines) { return y; } } else line = t;
  }
  if (line && n < maxLines) { g.fillText(line, x, y); y += lh; }
  return y;
}

const POSES = { e: { h: 4.5, y: 2.65 }, d: { h: 4.7, y: 2.7 }, g: { h: 4.4, y: 2.5 }, a: { h: 4.6, y: 2.6 } };
const VERT = 'uniform sampler2D dMap;uniform float amp;varying vec2 vUv;void main(){vUv=uv;float d=texture2D(dMap,uv).r;vec3 p=position;p.z+=(d-.3)*amp;gl_Position=projectionMatrix*modelViewMatrix*vec4(p,1.);}';
const FRAG = `uniform sampler2D map,dMap;uniform vec3 lightDir;uniform vec2 texel;uniform float alpha,flash;varying vec2 vUv;
void main(){vec4 c=texture2D(map,vUv);float a=smoothstep(.4,.9,c.a)*alpha;if(a<.01)discard;
 float dl=texture2D(dMap,vUv-vec2(texel.x*2.,0.)).r,dr=texture2D(dMap,vUv+vec2(texel.x*2.,0.)).r,du=texture2D(dMap,vUv+vec2(0.,texel.y*2.)).r,dd=texture2D(dMap,vUv-vec2(0.,texel.y*2.)).r;
 vec3 n=normalize(vec3((dl-dr)*8.,(dd-du)*8.,1.));vec3 L=normalize(lightDir);float diff=max(dot(n,L),0.);float sp=pow(max(dot(reflect(-L,n),vec3(0.,0.,1.)),0.),20.);
 vec3 col=c.rgb*(.74+.5*diff)+vec3(1.,.9,.8)*sp*.22;col*=vec3(1.05,1.0,.93);col+=vec3(1.,.3,.2)*flash*.12;gl_FragColor=vec4(col,a);}`;

    },

  };

  let raf, last = performance.now();
  function loop(now) {
    raf = requestAnimationFrame(loop); const dt = Math.min(.05, (now - last) / 1000); last = now; S.t += dt; const t = S.t;
    S.talk += (S.talkT - S.talk) * (1 - Math.exp(-dt * 18));

    // anchor motion: breathing, talking bob, lean forward when reading, mouse parallax

    poseGroup.position.y = Math.sin(t * 1.4) * .02 + S.talk * Math.abs(Math.sin(t * 9)) * .05;
    poseGroup.rotation.x = S.readingK * .06 + S.py * .03; poseGroup.rotation.y = S.px * .18 + Math.sin(t * .5) * .03 + S.talk * Math.sin(t * 3) * .05; poseGroup.rotation.z = Math.sin(t * .7) * .01 + S.talk * Math.sin(t * 6) * .012;
    poseGroup.scale.set(1 - S.talk * .01, 1 + S.talk * .02, 1);
    lightDir.set(-S.px * .9 + .3, -S.py * .9 + .5, 1);
    // newspaper: raised when reading, laid on the desk otherwise; page-flip animation
    const k = S.readingK; paper.position.set(0, 1.38 + (2.0 - 1.38) * k, 1.4); paper.rotation.x = -1.3 * (1 - k) - .1 * k;
    if (S.flip > 0) { S.flip -= dt * 2.2; const f = Math.max(S.flip, 0); paper.scale.x = Math.max(.02, Math.abs(Math.cos((1 - f) * Math.PI))); if (S.pendingPaper && f < .5) { paperTex.image = S.pendingPaper; paperTex.needsUpdate = true; S.pendingPaper = null; } if (f <= 0) paper.scale.x = 1; }
    onair.material.opacity = S.onair ? .7 + .3 * Math.abs(Math.sin(t * 3)) : .25; neon.material.opacity = .9 + Math.sin(t * 7) * .04 * (Math.random() > .97 ? 4 : 1);
    camera.position.x = S.px * .35; camera.lookAt(0, 3.1, 0);
    renderer.render(scene, camera);
  }
  raf = requestAnimationFrame(loop);
  return api;
}

const load = () => { try { const s = localStorage.getItem('spidey.news.cfg'); return s ? { ...DEF, ...JSON.parse(s), remember: true } : DEF; } catch (e) { return DEF; } };

      if (!quiet && fresh.length && seen.current.size > list.length) { studio.current?.setBreaking(true); setTimeout(() => studio.current?.setBreaking(false), 4000); }
      return list;
    } catch (e) { say(e.message); return []; }
  }

' + e.message); return templateBulletin(list); });

# Spidey News 24 – retro Spider-Man newsreader (React + Flutter)

Full explanation of every file, the AI (Groq LLM, Whisper, RAG), values and Flutter code is in the DOCUMENTATION Google-style doc shared in the chat.

## Fastest way (5 minutes, no coding)
1. Go to **app.netlify.com/drop** and drag the folder **READY-TO-DEPLOY-dist** onto the page. You get an https link. Open it on PC or phone.
2. Tap SETUP → paste your free Groq key (console.groq.com/keys) → DONE.
3. Tap **ON AIR** (Spidey reads the news) or **ASK** (ask by voice about the news).
Try `?demo=1` at the end of the link to see sample news without any internet feed.

## Edit / run the React code on your PC
Install Node.js (nodejs.org). In `react-app` run:  `npm install`  then  `npm run dev`  (opens http://localhost:5173).  Build for hosting: `npm run build` → creates `dist`.

## Live news every minute (recommended, free)
Deploy `worker/worker.js` on Cloudflare Workers (free): it merges BBC/Hindu/NDTV/Google News RSS + GDELT, caches 60 s, and hides your Groq key. Paste its URL into SETUP → "News Worker URL".
